const ID_LIVRO_MASTER = "price_1T1PskL12Q2zNNHlhCJqARTz"; async function comprarLivro() { const user = auth.currentUser; if(!user) return; const btnText = document.getElementById('livro-btn-text'); const originalText = btnText.innerText; btnText.innerText = "SINTONIZANDO..."; try { const response = await fetch('/api/create-checkout', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ priceId: ID_LIVRO_MASTER, uid: user.uid, email: user.email, mode: 'payment' }) }); // Estudo de Falhas: Verifica se o servidor respondeu com erro const data = await response.json(); if (!response.ok) { throw new Error(data.error || "Erro interno no servidor Vercel."); } if (data.url) { window.location.href = data.url; } else { throw new Error("URL de pagamento não gerada."); } } catch (error) { console.error("Erro técnico capturado:", error); // PRIORIDADE CINCO: O alerta agora mostrará a mensagem técnica real para correção cirúrgica alert("ALERTA TÉCNICO: " + error.message); btnText.innerText = "ADQUIRIR AGORA"; } }
